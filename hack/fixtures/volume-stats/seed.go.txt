// Built inside the pinned upstream driver module using its existing CSI/grpc
// dependencies. This test helper uses only the disposable driver's local socket.
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"time"

	"github.com/container-storage-interface/spec/lib/go/csi"
	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"
)

func main() {
	if err := create(); err != nil {
		fmt.Fprintln(os.Stderr, "fixture CSI CreateVolume failed")
		os.Exit(1)
	}
}

func create() error {
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	connection, err := grpc.NewClient("unix:///csi/csi.sock", grpc.WithTransportCredentials(insecure.NewCredentials()), grpc.WithDefaultCallOptions(grpc.MaxCallRecvMsgSize(64<<10)))
	if err != nil {
		return err
	}
	defer connection.Close()
	response, err := csi.NewControllerClient(connection).CreateVolume(ctx, &csi.CreateVolumeRequest{
		Name:          "volume-stats-fixture",
		CapacityRange: &csi.CapacityRange{RequiredBytes: 1 << 30},
		VolumeCapabilities: []*csi.VolumeCapability{{
			AccessType: &csi.VolumeCapability_Mount{Mount: &csi.VolumeCapability_MountVolume{}},
			AccessMode: &csi.VolumeCapability_AccessMode{Mode: csi.VolumeCapability_AccessMode_SINGLE_NODE_WRITER},
		}},
	})
	if err != nil {
		return err
	}
	if response.Volume == nil || response.Volume.VolumeId == "" {
		return fmt.Errorf("missing fixture identity")
	}
	return json.NewEncoder(os.Stdout).Encode(map[string]string{"volumeID": response.Volume.VolumeId})
}
